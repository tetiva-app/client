=== go ===
package main

import (
	"fmt"
	"strings"
	"net/http"
	"io"
)

func main() {

	url := "https://api.example.com/text"

	payload := strings.NewReader("\" + INJECTED + \" \\\" + INJECTED + \\\" ' . INJECTED . ' '; INJECTED; ' \\u0022 + INJECTED + \\u0022 ${INJECTED} {${INJECTED()}} $(INJECTED) `INJECTED` ?> */ \\ tab\there Привет\nline two end")

	req, _ := http.NewRequest("POST", url, payload)

	req.Header.Add("Content-Type", "text/plain; x=\"a\\\"b\"")

	res, _ := http.DefaultClient.Do(req)

	defer res.Body.Close()
	body, _ := io.ReadAll(res.Body)

	fmt.Println(string(body))

}
