=== go ===
package main

import (
	"fmt"
	"strings"
	"net/http"
	"io"
)

func main() {

	url := "https://api.example.com/users"

	payload := strings.NewReader("{\"name\":\"Ann\",\"age\":30,\"tags\":[\"x\",\"y\"],\"nested\":{\"ok\":true,\"n\":null}}")

	req, _ := http.NewRequest("POST", url, payload)

	req.Header.Add("Authorization", "Bearer abc")
	req.Header.Add("Content-Type", "application/json")

	res, _ := http.DefaultClient.Do(req)

	defer res.Body.Close()
	body, _ := io.ReadAll(res.Body)

	fmt.Println(string(body))

}

=== java-okhttp ===
OkHttpClient client = new OkHttpClient();

MediaType mediaType = MediaType.parse("application/json");
RequestBody body = RequestBody.create(mediaType, "{\"name\":\"Ann\",\"age\":30,\"tags\":[\"x\",\"y\"],\"nested\":{\"ok\":true,\"n\":null}}");
Request request = new Request.Builder()
  .url("https://api.example.com/users")
  .post(body)
  .addHeader("Authorization", "Bearer abc")
  .addHeader("Content-Type", "application/json")
  .build();

Response response = client.newCall(request).execute();

=== csharp-httpclient ===
using System.Net.Http.Headers;
var client = new HttpClient();
var request = new HttpRequestMessage
{
    Method = HttpMethod.Post,
    RequestUri = new Uri("https://api.example.com/users"),
    Headers =
    {
        { "Authorization", "Bearer abc" },
    },
    Content = new StringContent("{\"name\":\"Ann\",\"age\":30,\"tags\":[\"x\",\"y\"],\"nested\":{\"ok\":true,\"n\":null}}")
    {
        Headers =
        {
            ContentType = MediaTypeHeaderValue.Parse("application/json")
        }
    }
};
using (var response = await client.SendAsync(request))
{
    response.EnsureSuccessStatusCode();
    var body = await response.Content.ReadAsStringAsync();
    Console.WriteLine(body);
}
