=== python-requests ===
import requests

url = "https://api.example.com/users"
params = [
    ("existing", "1"),
    ("page", "2"),
    ("tag", "a b&c"),
    ("tag", "д"),
]
headers = {
    "Accept": "application/json",
}

response = requests.request("GET", url, params=params, headers=headers)
print(response.text)

=== go ===
package main

import (
	"fmt"
	"net/http"
	"io"
)

func main() {

	url := "https://api.example.com/users?existing=1&page=2&tag=a%20b%26c&tag=%D0%B4"

	req, _ := http.NewRequest("GET", url, nil)

	req.Header.Add("Accept", "application/json")

	res, _ := http.DefaultClient.Do(req)

	defer res.Body.Close()
	body, _ := io.ReadAll(res.Body)

	fmt.Println(string(body))

}

=== java-httpclient ===
HttpRequest request = HttpRequest.newBuilder()
    .uri(URI.create("https://api.example.com/users?existing=1&page=2&tag=a%20b%26c&tag=%D0%B4"))
    .header("Accept", "application/json")
    .method("GET", HttpRequest.BodyPublishers.noBody())
    .build();
HttpResponse<String> response = HttpClient.newHttpClient().send(request, HttpResponse.BodyHandlers.ofString());
System.out.println(response.body());

=== csharp-httpclient ===
using System.Net.Http.Headers;
var client = new HttpClient();
var request = new HttpRequestMessage
{
    Method = HttpMethod.Get,
    RequestUri = new Uri("https://api.example.com/users?existing=1&page=2&tag=a%20b%26c&tag=%D0%B4"),
    Headers =
    {
        { "Accept", "application/json" },
    },
};
using (var response = await client.SendAsync(request))
{
    response.EnsureSuccessStatusCode();
    var body = await response.Content.ReadAsStringAsync();
    Console.WriteLine(body);
}
