=== go ===
package main

import (
	"fmt"
	"strings"
	"net/http"
	"io"
)

func main() {

	url := "{{baseUrl}}/graphql"

	payload := strings.NewReader("{\"query\":\"query User($id: ID!) { user(id: $id) { name } }\",\"variables\":{\"id\":\"42\"},\"operationName\":\"User\"}")

	req, _ := http.NewRequest("POST", url, payload)

	req.Header.Add("Authorization", "Bearer {{token}}")
	req.Header.Add("Content-Type", "application/json")

	res, _ := http.DefaultClient.Do(req)

	defer res.Body.Close()
	body, _ := io.ReadAll(res.Body)

	fmt.Println(string(body))

}

=== csharp-httpclient ===
using System.Net.Http.Headers;
var client = new HttpClient();
var request = new HttpRequestMessage
{
    Method = HttpMethod.Post,
    RequestUri = new Uri("{{baseUrl}}/graphql"),
    Headers =
    {
        { "Authorization", "Bearer {{token}}" },
    },
    Content = new StringContent("{\"query\":\"query User($id: ID!) { user(id: $id) { name } }\",\"variables\":{\"id\":\"42\"},\"operationName\":\"User\"}")
    {
        Headers =
        {
            ContentType = MediaTypeHeaderValue.Parse("application/json")
        }
    }
};
using (var response = await client.SendAsync(request))
{
    response.EnsureSuccessStatusCode();
    var body = await response.Content.ReadAsStringAsync();
    Console.WriteLine(body);
}
