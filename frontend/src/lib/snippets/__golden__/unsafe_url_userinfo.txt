package main

import (
	"fmt"
	"net/http"
	"io"
)

func main() {

	url := "https://a'.INJECTED.'b%22c:p'w@api.example.com/p%27a%22t/h?q%27%22=%5Cv#f%27%22%5C"

	req, _ := http.NewRequest("GET", url, nil)

	res, _ := http.DefaultClient.Do(req)

	defer res.Body.Close()

	fmt.Println(string(body))

}
